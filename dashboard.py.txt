import streamlit as st
import pandas as pd
from strategy.signal_engine import generate_signal
from market.alpaca import get_market_data
from config.settings import WATCHLIST

st.set_page_config(layout="wide", page_title="News Trading LIVE")

st.title("?? News-Based Trading Dashboard (LIVE)")

data = []

for stock in WATCHLIST:
    signal = generate_signal(stock)
    price, volume, ma20, avg_volume = get_market_data(stock)

    data.append({
        "Stock": stock,
        "Price": round(price, 2),
        "Volume": volume,
        "MA20": round(ma20, 2),
        "Signal": signal or "NONE"
    })

df = pd.DataFrame(data)

st.dataframe(df, use_container_width=True)
